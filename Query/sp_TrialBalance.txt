ALTER PROCEDURE sp_TrialBalance  
(  
    @FromDate DATE,  
    @ToDate DATE,  
    @CompanyId INT,  
    @FinancialYearId INT  
)  
AS  
BEGIN  
    SET NOCOUNT ON;  
  
    ;WITH Accounts AS  
    (  
        -- All accounts that exist in transactions or annual closing OpeningBalance
        SELECT DISTINCT Accode  
        FROM GLTrans  
        WHERE CompanyId = @CompanyId  
          AND FinancialYearId = @FinancialYearId  
        UNION
        SELECT DISTINCT Accode
        FROM OpeningBalance
        WHERE CompanyId = @CompanyId
          AND FinancialYearId = @FinancialYearId
    ),  
  
    InitOpening AS
    (
        SELECT
            Accode,
            SUM(Debit - Credit) AS InitialBalance
        FROM OpeningBalance
        WHERE CompanyId = @CompanyId
          AND FinancialYearId = @FinancialYearId
        GROUP BY Accode
    ),

    PeriodOpening AS  
    (  
        SELECT   
            Accode,  
            SUM(Debit - Credit) AS PeriodBalance  
        FROM GLTrans  
        WHERE Docdate < @FromDate  
          AND CompanyId = @CompanyId  
          AND FinancialYearId = @FinancialYearId  
        GROUP BY Accode  
    ),  
  
    Period AS  
    (  
        SELECT   
            Accode,  
            SUM(Debit) AS Debit,  
            SUM(Credit) AS Credit  
        FROM GLTrans  
        WHERE Docdate BETWEEN @FromDate AND @ToDate  
          AND CompanyId = @CompanyId  
          AND FinancialYearId = @FinancialYearId  
        GROUP BY Accode  
    )  
  
    SELECT   
        A.Accode,  
        ISNULL(I.InitialBalance, 0) + ISNULL(O.PeriodBalance, 0) AS OpeningBalance,  
        ISNULL(P.Debit, 0) AS Debit,  
        ISNULL(P.Credit, 0) AS Credit,  
          
        (ISNULL(I.InitialBalance, 0) + ISNULL(O.PeriodBalance, 0)   
        + ISNULL(P.Debit, 0)   
        - ISNULL(P.Credit, 0)) AS ClosingBalance  
  
    FROM Accounts A  
    LEFT JOIN InitOpening I ON A.Accode = I.Accode
    LEFT JOIN PeriodOpening O ON A.Accode = O.Accode  
    LEFT JOIN Period P ON A.Accode = P.Accode  
  
    ORDER BY A.Accode;  
  
END
